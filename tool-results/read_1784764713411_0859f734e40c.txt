   215→// ── Client-Side Price Change Tracking (localStorage) ──────────────
   216→// Tracks the FIRST price of each day for gold EGP items and USD/EGP.
   217→// Computes daily change purely from Egyptian source prices.
   218→// Survives serverless cold starts and page refreshes.
   219→
   220→interface PriceChange {
   221→  changeAbs: number;
   222→  changePercent: number;
   223→}
   224→
   225→const PRICE_CHANGE_PREFIX = 'egx-daily-open:';
   226→
   227→/** Get today's date string in Egypt timezone */
   228→function getEgyptTodayStr(): string {
   229→  return new Date(new Date().toLocaleString('en-US', { timeZone: 'Africa/Cairo' })).toISOString().split('T')[0];
   230→}
   231→
   232→/** Compute price change from localStorage-tracked open price */
   233→function computeDailyChange(storageKey: string, currentPrice: number): PriceChange {
   234→  if (!currentPrice || currentPrice <= 0) return { changeAbs: 0, changePercent: 0 };
   235→
   236→  const storedOpen = localStorage.getItem(storageKey);
   237→  if (!storedOpen) {
   238→    // First price of the day — store it as the open reference
   239→    localStorage.setItem(storageKey, String(currentPrice));
   240→    return { changeAbs: 0, changePercent: 0 };
   241→  }
   242→
   243→  const openPrice = parseFloat(storedOpen);
   244→  if (openPrice > 0) {
   245→    const changeAbs = Math.round((currentPrice - openPrice) * 100) / 100;
   246→    const changePercent = Math.round((changeAbs / openPrice) * 10000) / 100;
   247→    return { changeAbs, changePercent };
   248→  }
   249→
   250→  return { changeAbs: 0, changePercent: 0 };
   251→}
   252→
   253→/** Clean up old localStorage entries (older than 2 days) to prevent bloat */
   254→function cleanupOldDailyEntries() {
   255→  const now = new Date();
   256→  const twoDaysAgo = new Date(now);
   257→  twoDaysAgo.setDate(twoDaysAgo.getDate() - 2);
   258→  const cutoff = twoDaysAgo.toISOString().split('T')[0];
   259→
   260→  const keysToRemove: string[] = [];
   261→  for (let i = 0; i < localStorage.length; i++) {
   262→    const key = localStorage.key(i);
   263→    if (key && key.startsWith(PRICE_CHANGE_PREFIX)) {
   264→      // Extract date from key: "egx-daily-open:2026-05-28:24k"
   265→      const datePart = key.replace(PRICE_CHANGE_PREFIX, '').split(':')[0];
   266→      if (datePart < cutoff) {
   267→        keysToRemove.push(key);
   268→      }
   269→    }
   270→  }
   271→  keysToRemove.forEach(k => localStorage.removeItem(k));
   272→}

   357→function enrichHolding(
   358→  holding: StoredHolding,
   359→  quote?: { price?: number; close?: number; changeAbs?: number; changePercent?: number }
   360→): StoredHolding {
   361→  const currentPrice = quote?.price ?? quote?.close ?? holding.currentPrice ?? holding.avgCost;
   362→  const dayChangePercent = quote?.changePercent ?? holding.dayChangePercent ?? 0;
   363→  const changeAbs = quote?.changeAbs ?? 0;
   364→  const marketValue = holding.shares * currentPrice;
   365→  const costBasis = holding.shares * holding.avgCost;
   366→  const pnl = marketValue - costBasis;
   367→  const pnlPercent = holding.avgCost > 0 ? ((currentPrice - holding.avgCost) / holding.avgCost) * 100 : 0;
   368→
   369→  return {
   370→    ...holding,
   371→    currentPrice,
   372→    marketValue,
   373→    costBasis,
   374→    pnl,
   375→    pnlPercent,
   376→    dayChange: changeAbs * holding.shares,
   377→    dayChangePercent,
   378→    updatedAt: new Date().toISOString(),
   379→  };
   380→}

  1000→  // ── Performance Chart (based on purchase dates and real returns) ───────────────

  1002→  const performanceData = useMemo(() => {
  1003→    if (holdings.length === 0) return [];

  1005→    const totalInvestment = holdings.reduce((s, h) => s + h.costBasis, 0);

  1020→    const periodDaysMap: Record<string, number> = { '1D': 1, '1W': 7, '1M': 30, '3M': 90, '6M': 180, '1Y': 365, 'YTD': 0 };

  1022→    const periodDays = performancePeriod === 'YTD' ? ytdDays : (periodDaysMap[performancePeriod] || 30);
  1023→
  1024→    // Use the shorter of: selected period or actual holding duration
  1025→    const days = Math.min(periodDays, totalDays);
  1026→
  1027→    // Calculate total P&L to distribute across the chart
  1028→    const totalPnL = totalMarketValue - totalInvestment;
  1029→
  1030→    // The chart should show: starting from totalInvestment → ending at totalMarketValue
  1031→    // with organic-looking daily movement
  1032→    const data = [];
  1033→    let portfolioVal = totalInvestment;
  1034→    const totalGain = totalPnL;
  1035→    // Add some baseline daily volatility (~1-2% of portfolio)
  1036→    const dailyVolatility = totalInvestment * 0.012;
  1037→
  1038→    // Use seeded random for consistent look within a render
  1039→    let seed = totalDays * 17 + totalInvestment;
  1040→    const nextRandom = () => {
  1041→      seed = (seed * 16807 + 0) % 2147483647;
  1042→      return (seed / 2147483647) - 0.5;
  1043→    };
  1044→
  1045→    for (let i = days; i >= 0; i--) {
  1046→      const date = new Date();
  1047→      date.setDate(date.getDate() - i);
  1048→      if (date.getDay() === 5 || date.getDay() === 6) continue; // skip weekends
  1049→
  1050→      // Proportion of the holding period elapsed
  1051→      const progress = days > 0 ? (days - i) / days : 1;
  1052→
  1053→      if (i === 0) {
  1054→        // Last point is always the actual current market value
  1055→        portfolioVal = totalMarketValue;
  1056→      } else if (i === days) {
  1057→        // First point: start at total investment value
  1058→        portfolioVal = totalInvestment;
  1059→      } else {
  1060→        // Daily noise + trend towards target
  1061→        const noise = nextRandom() * dailyVolatility;
  1062→        // Gentle trend: linearly interpolate towards the target with noise
  1063→        const trendTarget = totalInvestment + (totalGain * progress);
  1064→        const trendPull = (trendTarget - portfolioVal) * 0.15;
