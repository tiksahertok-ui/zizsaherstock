    56→function FadeIn({
    57→  children,
    58→  className,
    59→  delay = 0,
    60→  direction = 'up',
    61→  duration = 0.5,
    62→}: {
    63→  children: React.ReactNode;
    64→  className?: string;
    65→  delay?: number;
    66→  direction?: 'up' | 'down' | 'left' | 'right';
    67→  duration?: number;
    68→}) {
    69→  const dirMap = {
    70→    up: { y: 24, x: 0 },
    71→    down: { y: -24, x: 0 },
    72→    left: { x: 24, y: 0 },
    73→    right: { x: -24, y: 0 },
    74→  };
    75→  const offset = dirMap[direction];
    76→
    77→  return (
    78→    <motion.div
    79→      className={className}
    80→      initial={{ opacity: 0, ...offset }}
    81→      whileInView={{ opacity: 1, x: 0, y: 0 }}
    82→      viewport={{ once: true, margin: '-60px' }}
    83→      transition={{ duration, delay, ease: [0.21, 0.47, 0.32, 0.98] }}
    84→    >
    85→      {children}
    86→    </motion.div>
    87→  );
    88→}

    90→function StaggerContainer({ children, className }: { children: React.ReactNode; className?: string }) {

    97→      variants={{

   100→      }}
   101→    >
   102→      {children}
   103→    </motion.div>

   105→}

   116→// ── Counter Animation ─────────────────────────────────────────
   117→
   118→function AnimatedCounter({ target, suffix = '' }: { target: number; suffix?: string }) {
   119→  const [count, setCount] = useState(0);
   120→  const ref = React.useRef<HTMLSpanElement>(null);
   121→  const inView = useInView(ref, { once: true });
   122→
   123→  useEffect(() => {
   124→    if (!inView) return;
   125→    let start = 0;
   126→    const duration = 1500;
   127→    const step = (timestamp: number) => {
   128→      if (!start) start = timestamp;
   129→      const progress = Math.min((timestamp - start) / duration, 1);
   130→      const eased = 1 - Math.pow(1 - progress, 3);
   131→      setCount(Math.floor(eased * target));
   132→      if (progress < 1) requestAnimationFrame(step);
   133→    };
   134→    requestAnimationFrame(step);
   135→  }, [inView, target]);
   136→
   137→  return (
   138→    <span ref={ref}>
   139→      {count}
   140→      {suffix}
   141→    </span>
   142→  );
   143→}
