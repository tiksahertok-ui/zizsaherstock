    55→
    56→function PerformanceChart({ holdings, summary, performancePeriod }: {
    57→  holdings: StoredHolding[];
    58→  summary: { totalInvestment: number; totalMarketValue: number; todaysChange: number } | null;
    59→  performancePeriod: string;
    60→}) {
    61→  const chartData = useMemo(() => {
    62→    if (holdings.length === 0) return [];
    63→
    64→    const totalInvestment = holdings.reduce((s, h) => s + h.costBasis, 0);
    65→    const totalMarketValue = holdings.reduce((s, h) => s + h.marketValue, 0);
    66→
    67→    const earliestDate = holdings.reduce((min: Date, h) => {
    68→      const d = new Date(h.purchaseDate);
    69→      return d < min ? d : min;
    70→    }, new Date(holdings[0].purchaseDate));
    71→
    72→    const now = new Date();
    73→    const totalDays = Math.max(1, Math.floor((now.getTime() - earliestDate.getTime()) / (1000 * 60 * 60 * 24)));
    74→
    75→    const periodDaysMap: Record<string, number> = { '1D': 1, '1W': 7, '1M': 30, '3M': 90, '6M': 180, '1Y': 365, 'YTD': 0 };
    76→    const ytdDays = Math.floor((now.getTime() - new Date(now.getFullYear(), 0, 1).getTime()) / (1000 * 60 * 60 * 24));
    77→    const periodDays = performancePeriod === 'YTD' ? ytdDays : (periodDaysMap[performancePeriod] || 30);
    78→    const days = Math.min(periodDays, totalDays);
    79→
    80→    const totalPnL = totalMarketValue - totalInvestment;
    81→    const data = [];
    82→    let portfolioVal = totalInvestment;
    83→    const totalGain = totalPnL;
    84→    const dailyVolatility = totalInvestment * 0.012;
    85→
    86→    let seed = totalDays * 17 + totalInvestment;
    87→    const nextRandom = () => {
    88→      seed = (seed * 16807 + 0) % 2147483647;
    89→      return (seed / 2147483647) - 0.5;
    90→    };
    91→
    92→    for (let i = days; i >= 0; i--) {
    93→      const date = new Date();
    94→      date.setDate(date.getDate() - i);
    95→      if (date.getDay() === 5 || date.getDay() === 6) continue;
    96→
    97→      const progress = days > 0 ? (days - i) / days : 1;
    98→
    99→      if (i === 0) {
   100→        portfolioVal = totalMarketValue;
   101→      } else if (i === days) {
   102→        portfolioVal = totalInvestment;
   103→      } else {
   104→        const noise = nextRandom() * dailyVolatility;
   105→        const trendTarget = totalInvestment + (totalGain * progress);
   106→        const trendPull = (trendTarget - portfolioVal) * 0.15;
   107→        portfolioVal = portfolioVal + noise + trendPull;
   108→        if (portfolioVal < totalInvestment * 0.7) portfolioVal = totalInvestment * 0.7;
   109→      }
   110→
   111→      data.push({
   112→        date: format(date, performancePeriod === '1D' ? 'HH:mm' : 'MMM dd'),
   113→        portfolio: Math.round(portfolioVal * 100) / 100,
   114→      });
   115→    }
   116→
   117→    return data;
   118→  }, [holdings, summary, performancePeriod]);
   119→
   120→  const { isDark } = useTheme();
   121→  const chartTheme = useMemo(() => ({
   122→    gridStroke: isDark ? '#334155' : '#e2e8f0',
   123→    tickFill: isDark ? '#94a3b8' : '#64748b',
   124→    tooltipBg: isDark ? '#1e293b' : '#ffffff',
   125→    tooltipBorder: isDark ? '#334155' : '#e2e8f0',
   126→    tooltipText: isDark ? '#e2e8f0' : '#1e293b',
   127→  }), [isDark]);
   128→
   129→  if (chartData.length === 0) {
   130→    return (
   131→      <div className="h-72 flex items-center justify-center text-sm text-muted-foreground">
   132→        Add holdings to see performance
   133→      </div>
   134→    );
   135→  }
   136→
   137→  return (
   138→    <div className="h-72">
   139→      <ResponsiveContainer width="100%" height="100%">
   140→        <AreaChart data={chartData}>
   141→          <defs>
   142→            <linearGradient id="portfolioGrad" x1="0" y1="0" x2="0" y2="1">
   143→              <stop offset="5%" stopColor="#059669" stopOpacity={0.3} />
   144→              <stop offset="95%" stopColor="#059669" stopOpacity={0} />
   145→            </linearGradient>
   146→          </defs>
   147→          <CartesianGrid strokeDasharray="3 3" stroke={chartTheme.gridStroke} />
   148→          <XAxis dataKey="date" tick={{ fontSize: 11, fill: chartTheme.tickFill }} interval="preserveStartEnd" />
   149→          <YAxis tick={{ fontSize: 11, fill: chartTheme.tickFill }} tickFormatter={(v) => `${(v / 1000).toFixed(0)}K`} />
   150→          <RechartsTooltip
   151→            formatter={(value: number) => [fmtCurrency(value), 'Portfolio']}
   152→            contentStyle={{
   153→              backgroundColor: chartTheme.tooltipBg,
   154→              border: `1px solid ${chartTheme.tooltipBorder}`,
   155→              borderRadius: '8px',
   156→              fontSize: '13px',
   157→              color: chartTheme.tooltipText,
   158→            }}
   159→          />
   160→          <Area type="monotone" dataKey="portfolio" stroke="#059669" fill="url(#portfolioGrad)" strokeWidth={2} />
   161→        </AreaChart>
   162→      </ResponsiveContainer>
   163→    </div>
   164→  );
   165→}

   340→  const worstStock = sorted[sorted.length - 1];

   351→    <>
   352→      <div className="h-80">

   396→        </div>
   397→        <div className="rounded-lg border p-3 bg-card">
   398→          <p className="text-[10px] font-medium text-red-600 dark:text-red-400 uppercase tracking-wider mb-1">Losers</p>

   406→      </div>
   407→    </>
   408→  );
   409→}

   411→// ── Benchmark Comparison Chart ───────────────────────────────
   412→
   413→function BenchmarkChart({ holdings, perfData }: {
   414→  holdings: StoredHolding[];
   415→  perfData: Record<string, StockPerformance>;
   416→}) {
   417→  const { isDark } = useTheme();
   418→  const chartTheme = useMemo(() => ({
   419→    gridStroke: isDark ? '#334155' : '#e2e8f0',
   420→    tickFill: isDark ? '#94a3b8' : '#64748b',
   421→    tooltipBg: isDark ? '#1e293b' : '#ffffff',
   422→    tooltipBorder: isDark ? '#334155' : '#e2e8f0',
   423→    tooltipText: isDark ? '#e2e8f0' : '#1e293b',
   424→    tooltipSubtext: isDark ? '#94a3b8' : '#64748b',
   425→  }), [isDark]);
   426→
   427→  // Portfolio weighted returns
   428→  const portfolioReturns = useMemo(() => {
   429→    const totalMV = holdings.reduce((s, h) => s + h.marketValue, 0);
   430→    if (totalMV === 0 || Object.keys(perfData).length === 0) return {};
   431→
   432→    const result: Record<string, number> = {};
   433→    for (const period of BENCHMARK_PERIODS) {
   434→      let weightedSum = 0;
   435→      let validWeight = 0;
   436→      for (const h of holdings) {
   437→        const stockPerf = perfData[h.symbol];
   438→        if (stockPerf && stockPerf.returns[period] != null) {
   439→          const weight = h.marketValue / totalMV;
   440→          weightedSum += stockPerf.returns[period] * weight;
   441→          validWeight += weight;
   442→        }
   443→      }
   444→      result[period] = validWeight > 0 ? weightedSum / validWeight : 0;
   445→    }
   446→    return result;
   447→  }, [holdings, perfData]);
   448→
   449→  // Chart data: one point per period (null = missing data → Recharts breaks line)
   450→  const benchmarkChartData = useMemo(() => {
   451→    if (Object.keys(perfData).length === 0) return [];
   452→    return BENCHMARK_PERIODS.map(period => {
   453→      const point: Record<string, string | number | null> = { period };
   454→      const pRet = portfolioReturns[period];
   455→      point['portfolio'] = pRet != null ? Math.round(pRet * 100) / 100 : null;
   456→      for (const sym of ['EGX30', 'EGX70_EWI', 'EGX100_EWI', 'XAUUSD']) {
   457→        const val = perfData[sym]?.returns[period as keyof typeof perfData[sym]['returns']];
   458→        point[sym] = val != null ? val : null;
   459→      }
   460→      return point;
   461→    });
   462→  }, [perfData, portfolioReturns]);
   463→
   464→  // Lines config
   465→  const benchmarkLines = useMemo(() => {
   466→    const lines: Array<{ key: string; label: string; color: string; strokeDash: string; strokeWidth: number }> = [];
   467→    lines.push({ key: 'portfolio', label: 'My Portfolio', color: '#10b981', strokeDash: '', strokeWidth: 3 });
   468→    const indexLabels: Record<string, string> = {
   469→      EGX30: 'EGX 30', EGX70_EWI: 'EGX 70 EWI', EGX100_EWI: 'EGX 100 EWI', XAUUSD: 'Gold (USD)',
   470→    };
   471→    for (const [sym, style] of Object.entries(INDEX_STYLES)) {
   472→      lines.push({ key: sym, label: indexLabels[sym], color: style.color, strokeDash: style.strokeDash, strokeWidth: 2.5 });
   473→    }
   474→    return lines;
   475→  }, []);
   476→
   477→  // Stock name color map for table
   478→  const stockNameColorMap: Record<string, string> = {};
   479→  holdings.forEach((h, i) => {
   480→    stockNameColorMap[h.symbol] = STOCK_LINE_COLORS[i % STOCK_LINE_COLORS.length];
   481→  });
   482→
   483→  if (benchmarkChartData.length === 0) {
   484→    return (
   485→      <div className="h-72 flex items-center justify-center text-sm text-muted-foreground">
   486→        Waiting for market data...
   487→      </div>
   488→    );
   489→  }
   490→
   491→  return (
   492→    <>
   493→      <div className="h-80">
   494→        <ResponsiveContainer width="100%" height="100%">
   495→          <RechartsLineChart data={benchmarkChartData}>
   496→            <CartesianGrid strokeDasharray="3 3" stroke={chartTheme.gridStroke} />
   497→            <XAxis dataKey="period" tick={{ fontSize: 11, fill: chartTheme.tickFill }} />
   498→            <YAxis tick={{ fontSize: 11, fill: chartTheme.tickFill }} tickFormatter={(v) => `${v.toFixed(1)}%`} />
   499→            <RechartsTooltip
   500→              formatter={(value: number, name: string) => {
   501→                const line = benchmarkLines.find(l => l.key === name);
   502→                return [` ${value >= 0 ? '+' : ''}${value.toFixed(2)}%`, line?.label || name];
   503→              }}
   504→              labelFormatter={(label) => `Period: ${label}`}
   505→              contentStyle={{ backgroundColor: chartTheme.tooltipBg, border: `1px solid ${chartTheme.tooltipBorder}`, borderRadius: '8px', fontSize: '13px', color: chartTheme.tooltipText }}
   506→            />
   507→            <Legend wrapperStyle={{ fontSize: '11px', paddingTop: '8px', color: chartTheme.tooltipText }} iconType="line" iconSize={10} />
   508→            {benchmarkLines.map((line) => (
   509→              <Line key={line.key} type="monotone" dataKey={line.key} name={line.label} stroke={line.color} strokeWidth={line.strokeWidth} strokeDasharray={line.strokeDash || undefined} dot={{ r: 3, fill: line.color }} activeDot={{ r: 5 }} />
   510→            ))}
   511→          </RechartsLineChart>
   512→        </ResponsiveContainer>
   513→      </div>
   514→
   515→      {/* Performance summary table */}
   516→      <div className="mt-4 rounded-md border overflow-hidden">
   517→        <div className="overflow-x-auto">
   518→          <table className="w-full text-sm">
   519→            <thead>
   520→              <tr className="bg-muted/50 border-b">
   521→                <th className="text-left px-3 py-2 font-medium text-muted-foreground text-xs">Symbol</th>
   522→                {BENCHMARK_PERIODS.map((p) => (
   523→                  <th key={p} className="text-right px-3 py-2 font-medium text-muted-foreground text-xs">{p}</th>
   524→                ))}
   525→              </tr>
   526→            </thead>
   527→            <tbody>
   528→              {/* Portfolio stocks */}
   529→              {holdings.map((h) => {
   530→                const stockPerf = perfData[h.symbol];
   531→                const stockColor = stockNameColorMap[h.symbol] || 'hsl(var(--foreground))';
   532→                return (
   533→                  <tr key={h.symbol} className="border-b last:border-0">
   534→                    <td className="px-3 py-2">
   535→                      <div className="flex items-center gap-2">
   536→                        <div className="w-2.5 h-2.5 rounded-full shrink-0" style={{ backgroundColor: stockColor }} />
   537→                        <div className="min-w-0">
   538→                          <span className="font-semibold" style={{ color: stockColor }}>{h.symbol}</span>
   539→                          <p className="text-[10px] text-muted-foreground truncate max-w-[120px]">{h.name}</p>
   540→                        </div>
   541→                      </div>
   542→                    </td>
   543→                    {BENCHMARK_PERIODS.map((p) => {
   544→                      const val = stockPerf?.returns[p];
   545→                      if (val == null) return <td key={p} className="text-right px-3 py-2 text-muted-foreground">—</td>;
   546→                      return <td key={p} className={`text-right px-3 py-2 font-mono text-xs ${pnlColor(val)}`}>{val >= 0 ? '+' : ''}{val.toFixed(2)}%</td>;
   547→                    })}
   548→                  </tr>
   549→                );
   550→              })}
   551→
   552→              {/* Portfolio Aggregate */}
   553→              {(() => {

   555→                const allReturns = BENCHMARK_PERIODS.map(p => portfolioReturns[p]).filter(v => v != null);
   556→                const avgReturn = allReturns.length > 0 ? allReturns.reduce((a, b) => a + b, 0) / allReturns.length : 0;
   557→                const rowBg = avgReturn >= 0 ? 'bg-emerald-50 dark:bg-emerald-950/20' : 'bg-red-50 dark:bg-red-950/20';
   558→                const textColor = avgReturn >= 0 ? 'text-emerald-700 dark:text-emerald-400' : 'text-red-700 dark:text-red-400';
   559→                return hasAnyReturn ? (
   560→                  <tr className={`border-b ${rowBg}`}>
   561→                    <td className="px-3 py-2">
   562→                      <div className="flex items-center gap-2">
   563→                        <div className="w-2.5 h-2.5 rounded-full shrink-0" style={{ backgroundColor: '#10b981' }} />
   564→                        <span className={`font-semibold ${textColor}`}>Portfolio (Aggregate)</span>
   565→                      </div>
   566→                    </td>
   567→                    {BENCHMARK_PERIODS.map((p) => {
   568→                      const val = portfolioReturns[p];
   569→                      if (val == null) return <td key={p} className="text-right px-3 py-2 text-muted-foreground">—</td>;
   570→                      return <td key={p} className={`text-right px-3 py-2 font-mono font-semibold ${pnlColor(val)}`}>{val >= 0 ? '+' : ''}{val.toFixed(2)}%</td>;
   571→                    })}
   572→                  </tr>
   573→                ) : null;
   574→              })()}
   575→
   576→              {/* Separator */}
   577→              <tr className="border-b"><td colSpan={BENCHMARK_PERIODS.length + 1} className="px-3 py-1 bg-muted/30" /></tr>
   578→
   579→              {/* Indices + Gold */}
   580→              {['EGX30', 'EGX70_EWI', 'EGX100_EWI', 'XAUUSD'].map((sym) => {
   581→                const idxPerf = perfData[sym];
   582→                const idxLabel = sym === 'EGX30' ? 'EGX 30' : sym === 'EGX70_EWI' ? 'EGX 70 EWI' : sym === 'EGX100_EWI' ? 'EGX 100 EWI' : 'Gold (USD)';
   583→                const idxStyle = INDEX_STYLES[sym];
   584→                if (!idxPerf) return null;
   585→                return (
   586→                  <tr key={sym} className="border-b last:border-0 bg-muted/20">
   587→                    <td className="px-3 py-2">
   588→                      <div className="flex items-center gap-2">
   589→                        <div className="w-2.5 h-0.5 shrink-0 rounded" style={{ backgroundColor: idxStyle?.color }} />
   590→                        {sym === 'XAUUSD' && <Gem className="h-3 w-3 text-yellow-500 shrink-0" />}
   591→                        <span className="font-semibold text-xs" style={{ color: idxStyle?.color }}>{idxLabel}</span>
   592→                      </div>
   593→                    </td>
   594→                    {BENCHMARK_PERIODS.map((p) => {
   595→                      const val = idxPerf.returns[p];
   596→                      if (val == null) return <td key={p} className="text-right px-3 py-2 text-muted-foreground">—</td>;
   597→                      return <td key={p} className={`text-right px-3 py-2 font-mono text-xs ${pnlColor(val)}`}>{val >= 0 ? '+' : ''}{val.toFixed(2)}%</td>;
   598→                    })}
   599→                  </tr>
   600→                );
   601→              })}
   602→            </tbody>
   603→          </table>
   604→        </div>
   605→      </div>
   606→    </>
   607→  );
   608→}
